for win

python -m venv .venv  установка venv
.venv\Scripts\python -m pip install -r requirements.txt

.venv\Scripts\python -m uvicorn app.main:app --reload запуск сервера

.venv\Scripts\python -m pytest                запуск unit тесты
.venv\Scripts\python -m pytest tests/api -v   запуск api тесты

for mac os

python3 -m venv .venv
.venv/bin/python -m pip install -r requirements.txt

.venv/bin/python -m uvicorn app.main:app   запуск сервера

.venv/bin/python -m pytest tests/unit -v  запуск unit тесты
.venv/bin/python -m pytest tests/api -v  запуск api тесты
